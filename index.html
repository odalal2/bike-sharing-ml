# Run when hour.csv is imported\nbike <- read.csv("hour.csv")\nfeatures <- c("yr","mnth","hr","holiday","weekday","workingday","weathersit","temp","hum","windspeed","season")\nbike <- bike[order(as.Date(bike$dteday), bike$hr), ]\nmodel_data <- bike[complete.cases(bike[,c(features,"cnt")]), c("dteday",features,"cnt")]\nsplit <- floor(0.8 * nrow(model_data))\ntrain <- model_data[1:split, ]\ntest <- model_data[(split+1):nrow(model_data), ]
